  <script>
    /* 1. مقاطع القصائد المأخوذة من الصفحات المرفقة */
    const poems = [
      {
        title: "من قصيدة: للعيد أبو هلالين",
        text: "بعدها امحبتج خضره\nذبل عمري ولا ذبلت\nيجمره وشابكه اعله الروح\nلا تبرد ... ولا تهفت"
      },
      {
        title: "من قصيدة: للعيد أبو هلالين",
        text: "يبعيده الشراع الضاع\nبيه ريح الهجر عصفت"
      },
      {
        title: "من قصيدة: هواجس ليليه",
        text: "آنه شبيدي .. آنه شبيدي\nيها ........ يالرايحه امن ايدي\nيماي وطاح مني ابكاع عطشانه\nوبكه ايلالي العطش بيدي"
      },
      {
        title: "من قصيدة: هواجس ليليه",
        text: "يصيده وعالجرف ضاعت\nاكل عمري البحر لمن وصلتلهه\nورجع يبجي الشبح بيدي\nوبكه ايلوح الحزن بيدي"
      },
      {
        title: "من قصيدة: ربيع اذار",
        text: "غني .... ديرتنه ابربيع اذار\nمر بيه الحام\nغني .... غني\nغني كل فرحه زمانك\nللفرح .... للناس .... للحب\nللنخيل الدوم فاتح كلبه لطيور السلام"
      },
      {
        title: "من قصيدة: ربيع اذار",
        text: "غني .... للنسمات .... للكاع الحبيبه\nواظفر اخيوط الشمس ....\nاظفر اخيوط الشمس لجبالنه الحلوه كصيبه"
      }
    ];

    let currentPoem = 0;
    const poemBox = document.getElementById("poemBox");
    const poemTitle = document.getElementById("poemTitle");
    const poemLines = document.getElementById("poemLines");
    const nextPoemBtn = document.getElementById("nextPoemBtn");
    const walker = document.getElementById("walker");

    function showNextPoem() {
      poemBox.classList.add("fade");
      setTimeout(() => {
        currentPoem = (currentPoem + 1) % poems.length;
        poemTitle.textContent = poems[currentPoem].title;
        poemLines.textContent = poems[currentPoem].text;
        poemBox.classList.remove("fade");
      }, 350);
    }

    nextPoemBtn.addEventListener("click", (e) => {
      e.stopPropagation();
      showNextPoem();
      startMusicIfNeeded();
    });

    walker.addEventListener("click", (e) => {
      e.stopPropagation();
      showNextPoem();
      startMusicIfNeeded();
    });

    // تبديل تلقائي للأبيات كل 9 ثوانٍ
    setInterval(showNextPoem, 9000);

    /* 2. حركة مشي الشخصية ذهاباً وإياباً */
    let posX = 40;
    let direction = 1; // 1 لليمين، -1 لليسار
    const speed = 1.25;

    function animateWalker() {
      const maxWidth = window.innerWidth - 110;
      posX += speed * direction;

      if (posX >= maxWidth) {
        posX = maxWidth;
        direction = -1;
        walker.classList.add("flip");
      } else if (posX <= 20) {
        posX = 20;
        direction = 1;
        walker.classList.remove("flip");
      }

      walker.style.left = posX + "px";
      requestAnimationFrame(animateWalker);
    }
    animateWalker();

    /* 3. تأثير المطر على الشاشة */
    const canvas = document.getElementById("rainCanvas");
    const ctx = canvas.getContext("2d");
    let drops = [];

    function resizeCanvas() {
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;
    }
    window.addEventListener("resize", resizeCanvas);
    resizeCanvas();

    for (let i = 0; i < 160; i++) {
      drops.push({
        x: Math.random() * canvas.width,
        y: Math.random() * canvas.height,
        len: 12 + Math.random() * 18,
        speed: 6 + Math.random() * 7,
        opacity: 0.2 + Math.random() * 0.45
      });
    }

    function drawRain() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      ctx.lineWidth = 1.2;

      for (const d of drops) {
        ctx.strokeStyle = `rgba(203, 213, 225, ${d.opacity})`;
        ctx.beginPath();
        ctx.moveTo(d.x, d.y);
        ctx.lineTo(d.x - 2.5, d.y + d.len);
        ctx.stroke();

        d.y += d.speed;
        d.x -= 1.2;

        if (d.y > canvas.height) {
          d.y = -20;
          d.x = Math.random() * (canvas.width + 50);
        }
      }
      requestAnimationFrame(drawRain);
    }
    drawRain();

    /* 4. موسيقى درامية حزينة مولدة برمجياً (بيانو خفيف + صوت مطر ناعم) */
    let audioCtx = null;
    let isPlaying = false;
    let melodyTimer = null;
    const musicBtn = document.getElementById("musicBtn");

    // ترددات نغمات درامية حزينة (سلم لا الصغير A Minor)
    const chords = [
      [220.00, 261.63, 329.63], // Am
      [174.61, 220.00, 261.63], // F
      [130.81, 196.00, 261.63], // C
      [164.81, 246.94, 329.63]  // Em
    ];
    let chordStep = 0;

    function playPianoNote(freq, duration = 2.4) {
      if (!audioCtx || !isPlaying) return;
      const now = audioCtx.currentTime;
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();

      osc.type = "triangle";
      osc.frequency.setValueAtTime(freq, now);

      gain.gain.setValueAtTime(0.0001, now);
      gain.gain.exponentialRampToValueAtTime(0.06, now + 0.08);
      gain.gain.exponentialRampToValueAtTime(0.0001, now + duration);

      osc.connect(gain);
      gain.connect(audioCtx.destination);

      osc.start(now);
      osc.stop(now + duration);
    }

    function startDramaticMusic() {
      if (!audioCtx) {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      }
      if (audioCtx.state === "suspended") {
        audioCtx.resume();
      }
      isPlaying = true;
      musicBtn.textContent = "إيقاف الموسيقى ⏸";
      musicBtn.classList.add("active");

      const playStep = () => {
        if (!isPlaying) return;
        const chord = chords[chordStep % chords.length];
        playPianoNote(chord[0] / 2, 3.0); // نغمة باس عميقة
        setTimeout(() => playPianoNote(chord[0], 2.2), 250);
        setTimeout(() => playPianoNote(chord[1], 2.2), 650);
        setTimeout(() => playPianoNote(chord[2], 2.4), 1050);
        chordStep++;
      };

      playStep();
      melodyTimer = setInterval(playStep, 2600);
    }

    function stopDramaticMusic() {
      isPlaying = false;
      clearInterval(melodyTimer);
      musicBtn.textContent = "تشغيل الموسيقى الدرامية 🎵";
      musicBtn.classList.remove("active");
    }

    function startMusicIfNeeded() {
      if (!isPlaying) startDramaticMusic();
    }

    musicBtn.addEventListener("click", (e) => {
      e.stopPropagation();
      if (isPlaying) {
        stopDramaticMusic();
      } else {
        startDramaticMusic();
      }
    });
  </script>
